884: bool PlayerbotAIConfig::IsInRandomAccountList(uint32 id)
885: {
886:     // Fast path: already confirmed bot account.
887:     if (find(randomBotAccounts.begin(), randomBotAccounts.end(), id) != randomBotAccounts.end())
888:         return true;
889:     // Fast path: already confirmed non-bot account — skip the DB query.
890:     if (nonRandomBotAccounts.count(id))
891:         return false;
892: 
893:     // Slow path: look up username and check RNDBOT prefix. Cache result either way.
894:     auto qr = LoginDatabase.PQuery("SELECT username FROM account WHERE id = %u", id);
895:     if (!qr)
896:     {
897:         nonRandomBotAccounts.insert(id);
898:         return false;
899:     }
900:     Field* fields = qr->Fetch();
901:     std::string username = fields[0].GetCppString();
902:     std::string prefix = randomBotAccountPrefix;
903:     bool isBot = username.size() >= prefix.size();
904:     for (size_t i = 0; isBot && i < prefix.size(); ++i)
905:         isBot = std::tolower((unsigned char)username[i]) == std::tolower((unsigned char)prefix[i]);
906: 
907:     if (isBot)
908:         randomBotAccounts.push_back(id);
909:     else
910:         nonRandomBotAccounts.insert(id);
911:     return isBot;
