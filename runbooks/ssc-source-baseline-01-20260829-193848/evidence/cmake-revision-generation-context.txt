   338: #)
   339: 
   340: # Find core revision
   341: if(NOT GIT_EXECUTABLE)
   342:   set(rev_date "1970-01-01 00:00:00 +0000")
   343:   set(rev_hash "unknown")
   344: else()
   345:   execute_process(
   346:     COMMAND ${GIT_EXECUTABLE} rev-parse --short=20 HEAD
   347:     WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
   348:     OUTPUT_VARIABLE rev_hash
   349:     OUTPUT_STRIP_TRAILING_WHITESPACE
   350:     ERROR_QUIET
   351:   )
   352: 
   353:   execute_process(
   354:     COMMAND ${GIT_EXECUTABLE} show -s --format=%ci
   355:     WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
   356:     OUTPUT_VARIABLE rev_date
   357:     OUTPUT_STRIP_TRAILING_WHITESPACE
   358:     ERROR_QUIET
   359:   )
   360:   
   361:   if(NOT rev_hash)
   362:     # No valid ways available to find/set the revision/hash, so let's force some defaults
   363:     message(STATUS "
   364:     Could not find a proper repository signature (hash) - you may need to pull tags with git fetch -t
   365:     Continuing anyway - note that the versionstring will be set to \"unknown 1970-01-01 00:00:00 (Archived)\"")
   366:     set(rev_date "1970-01-01 00:00:00 +0000")
   367:     set(rev_hash "unknown")
   368:   endif()
   369:   
   370:   configure_file(${CMAKE_CURRENT_SOURCE_DIR}/cmake/revision.h.cmake ${CMAKE_CURRENT_SOURCE_DIR}/src/shared/revision.h)
   371: endif()
